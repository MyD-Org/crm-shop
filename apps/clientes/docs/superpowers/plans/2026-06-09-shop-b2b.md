# Shop B2B — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Tienda online integrada con Flexxus donde cada cliente ve sus precios según la lista asignada, con precios dinámicos por cantidad y carrito de compras.

**Architecture:** Next.js 15 App Router. Misma app que el portal de clientes (route group `(shop)`). La lista de precios asignada al cliente se lee desde su perfil en Flexxus (`GET /clientes/{codigo}`) y se almacena en la sesión existente (iron-session). El catálogo se consume desde Flexxus en tiempo real. Los precios por cantidad se consultan en Flexxus via `/articulos/{codigo}/precio` con parámetro `codigolista` y cantidad. PostgreSQL solo almacena datos que Flexxus no gestiona (fotos adicionales, descripción web, peso/dimensiones).

**Tech Stack:** Next.js 15, TypeScript, Tailwind CSS 4, Prisma + PostgreSQL, iron-session (sesión existente del portal), Flexxus API v5.

> **Prerequisito:** Este plan requiere que el Portal de Clientes (plan `2026-06-08-portal-clientes.md`) esté implementado, ya que reutiliza auth, sesión y el cliente Flexxus.

> **Pendiente confirmar con cliente:** Nombre exacto del campo `codigolista` en el objeto cliente de Flexxus. Verificar con `GET /clientes/{codigo}` en el swagger antes de implementar Task 3.

```typescript
export async function getProductos(params?: {
  busqueda?: string
  codigorubro?: string
  codigomarca?: string
}) {
  return apiFetch<any[]>('/listadoarticulos', params ?? {})

export async function getProducto(codigoarticulo: string) {
  return apiFetch<any>(`/productos/${codigoarticulo}`)
}

export async function getPrecioConLista(codigoarticulo: string, codigolista: string) {
  return apiFetch<any>(`/articulos/${codigoarticulo}/precio`, { codigolista })
}

export async function getPreciosPorCantidad(codigoarticulo: string, codigolista: string): Promise<Array<{
  cantidaddesde: number
  cantidadhasta: number | null
  precio: number
}>> {
  // Flexxus puede manejar escalas de precio via parámetro "escala" o similar.
  // Verificar endpoint exacto en swagger: /articulos/{codigo}/precio?codigolista=X&escala=true
  // Si no existe endpoint de escala, este array se construye con múltiples consultas de precio por cantidad
  try {
    return await apiFetch<any[]>(`/articulos/${codigoarticulo}/precio`, {
      codigolista,
      escala: 'true',
    })
  } catch {
    // fallback: retornar precio único sin escala
    const precio = await getPrecioConLista(codigoarticulo, codigolista)
    return [{ cantidaddesde: 1, cantidadhasta: null, precio: precio.precio }]
  }
}
```

```typescript
// src/app/api/shop/productos/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { getSession } from '@/lib/session'
import { getProductos } from '@/lib/flexxus'
import { prisma } from '@/lib/prisma'

export async function GET(req: NextRequest) {
  const session = await getSession()
  if (!session.codigocliente) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

  const { searchParams } = new URL(req.url)
  const productos = await getProductos({
    busqueda: searchParams.get('busqueda') ?? undefined,
    codigorubro: searchParams.get('rubro') ?? undefined,
    codigomarca: searchParams.get('marca') ?? undefined,
  })

  const codigosArticulo = productos.map((p: any) => p.codigoarticulo)
  const extras = await prisma.productoExtra.findMany({
    where: { codigoarticulo: { in: codigosArticulo }, activo: true },
  })
  const extrasMap = Object.fromEntries(extras.map((e) => [e.codigoarticulo, e]))

  const resultado = productos.map((p: any) => ({
    ...p,
    extra: extrasMap[p.codigoarticulo] ?? null,
  }))

  return NextResponse.json(resultado)
}
```

```typescript
// src/app/api/shop/carrito/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { getSession } from '@/lib/session'
import { prisma } from '@/lib/prisma'

export async function GET(req: NextRequest) {
  const session = await getSession()
  if (!session.codigocliente) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

  const items = await prisma.carritoItem.findMany({
    where: { sessionId: session.codigocliente },
    orderBy: { createdAt: 'asc' },
  })
  return NextResponse.json(items)
}

export async function POST(req: NextRequest) {
  const session = await getSession()
  if (!session.codigocliente) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

  const { codigoarticulo, cantidad } = await req.json()

  const existing = await prisma.carritoItem.findFirst({
    where: { sessionId: session.codigocliente, codigoarticulo },
  })

  if (existing) {
    const updated = await prisma.carritoItem.update({
      where: { id: existing.id },
      data: { cantidad },
    })
    return NextResponse.json(updated)
  }

  const item = await prisma.carritoItem.create({
    data: { sessionId: session.codigocliente, codigoarticulo, cantidad },
  })
  return NextResponse.json(item, { status: 201 })
}

export async function DELETE(req: NextRequest) {
  const session = await getSession()
  if (!session.codigocliente) return NextResponse.json({ error: 'No autorizado' }, { status: 401 })

  const { codigoarticulo } = await req.json()

  await prisma.carritoItem.deleteMany({
    where: { sessionId: session.codigocliente, codigoarticulo },
  })
  return NextResponse.json({ ok: true })
}
```

```tsx
// src/components/shop/PrecioTabla.tsx
'use client'

interface Escala {
  cantidaddesde: number
  cantidadhasta: number | null
  precio: number
}

export function PrecioTabla({
  escalas,
  cantidadSeleccionada,
}: {
  escalas: Escala[]
  cantidadSeleccionada: number
}) {
  const fmt = (n: number) =>
    new Intl.NumberFormat('es-AR', { style: 'currency', currency: 'ARS' }).format(n)

  const precioActual = escalas.findLast(
    (e) => cantidadSeleccionada >= e.cantidaddesde
  )?.precio ?? escalas[0]?.precio

  if (escalas.length <= 1) {
    return (
      <p className="text-2xl font-semibold text-gray-900">
        {fmt(escalas[0]?.precio ?? 0)}
      </p>
    )
  }

  return (
    <div className="space-y-2">
      <p className="text-2xl font-semibold text-gray-900">{fmt(precioActual)}</p>
      <div className="border rounded-lg overflow-hidden text-sm">
        <table className="w-full">
          <thead className="bg-gray-50">
            <tr>
              <th className="px-4 py-2 text-left text-xs text-gray-500 font-medium">Cantidad</th>
              <th className="px-4 py-2 text-right text-xs text-gray-500 font-medium">Precio unit.</th>
            </tr>
          </thead>
          <tbody>
            {escalas.map((e, i) => {
              const activa =
                cantidadSeleccionada >= e.cantidaddesde &&
                (e.cantidadhasta === null || cantidadSeleccionada <= e.cantidadhasta)
              return (
                <tr
                  key={i}
                  className={`border-t ${activa ? 'bg-blue-50 font-medium' : ''}`}
                >
                  <td className="px-4 py-2">
                    {e.cantidadhasta === null
                      ? `${e.cantidaddesde}+`
                      : `${e.cantidaddesde} – ${e.cantidadhasta}`}
                  </td>
                  <td className="px-4 py-2 text-right">{fmt(e.precio)}</td>
                </tr>
              )
            })}
          </tbody>
        </table>
      </div>
    </div>
  )
}
```

```tsx
// src/app/(shop)/carrito/page.tsx
'use client'
import { useEffect, useState } from 'react'
import { CarritoItem } from '@/components/shop/CarritoItem'

export default function CarritoPage() {
  const [items, setItems] = useState<any[]>([])
  const [precios, setPrecios] = useState<Record<string, number>>({})

  async function cargarCarrito() {
    const res = await fetch('/api/shop/carrito')
    const data = await res.json()
    setItems(data)

    const preciosNuevos: Record<string, number> = {}
    await Promise.all(
      data.map(async (item: any) => {
        const res = await fetch(`/api/shop/productos/${item.codigoarticulo}`)
        const prod = await res.json()
        const precio = prod.escalas?.findLast(
          (e: any) => item.cantidad >= e.cantidaddesde
        )?.precio ?? prod.escalas?.[0]?.precio ?? 0
        preciosNuevos[item.codigoarticulo] = precio
      })
    )
    setPrecios(preciosNuevos)
  }

  useEffect(() => { cargarCarrito() }, [])

  async function actualizarCantidad(codigoarticulo: string, cantidad: number) {
    await fetch('/api/shop/carrito', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ codigoarticulo, cantidad }),
    })
    cargarCarrito()
  }

  async function eliminarItem(codigoarticulo: string) {
    await fetch('/api/shop/carrito', {
      method: 'DELETE',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ codigoarticulo }),
    })
    cargarCarrito()
  }

  const total = items.reduce(
    (acc, item) => acc + (precios[item.codigoarticulo] ?? 0) * item.cantidad,
    0
  )

  return (
    <div className="max-w-2xl mx-auto space-y-6">
      <h1 className="text-2xl font-semibold">Carrito</h1>

      {items.length === 0 ? (
        <div className="bg-white border rounded-xl p-12 text-center">
          <p className="text-gray-500">Tu carrito está vacío</p>
        </div>
      ) : (
        <div className="bg-white border rounded-xl px-6">
          {items.map((item) => (
            <CarritoItem
              key={item.id}
              item={item}
              precio={precios[item.codigoarticulo] ?? 0}
              onCantidadChange={(c) => actualizarCantidad(item.codigoarticulo, c)}
              onEliminar={() => eliminarItem(item.codigoarticulo)}
            />
          ))}
          <div className="border-t py-4 flex justify-between items-center">
            <span className="font-semibold">Total</span>
            <span className="text-xl font-semibold">
              {new Intl.NumberFormat('es-AR', { style: 'currency', currency: 'ARS' }).format(total)}
            </span>
          </div>
          <div className="pb-6">
            <button
              className="w-full bg-blue-600 text-white py-3 rounded-lg font-medium hover:bg-blue-700"
              onClick={() => alert('Checkout — Fase 3: Pasarela de pago')}
            >
              Confirmar pedido
            </button>
          </div>
        </div>
      )}
    </div>
  )
}
```
